# Builds ns-3 contrib modules (the org's ns-3 modules: routing-bench,
# swarm-adapter-ns3's swarm-core) against the ns3 port's installed ns-3,
# outside an ns-3 tree. The ns3 port installs this project as
# share/ns3/contrib/; a module port configures it:
#
#   vcpkg_cmake_configure(SOURCE_PATH "${CURRENT_INSTALLED_DIR}/share/ns3/contrib"
#       OPTIONS -DNS3_CONTRIB_PACKAGE=${PORT} "-DNS3_CONTRIB_MODULES=<dir>;<dir>")
#
# It stands in for what ns-3's own build gives a contrib/ module: the
# build_lib() and build_lib_example() calls, the ${lib<module>} variables of
# the installed modules, NS3_VER and CMAKE_HEADER_OUTPUT_DIRECTORY. Each module
# becomes the shared library libns<NS3_VER>-<module>[-debug].so (ns-3's
# naming), its headers ns3/<header>.h with the umbrella ns3/<module>-module.h,
# its examples programs in bin/, and the package <NS3_CONTRIB_PACKAGE>
# exports the targets ns3::<module>, as the ns3 package names its own. Test sources
# are not built: they need ns-3's test runner, which the ns3 port leaves out.
cmake_minimum_required(VERSION 3.21)
project(ns3-contrib LANGUAGES CXX)

set(NS3_CONTRIB_PACKAGE "" CACHE STRING "the CMake package to install (the vcpkg port's name)")
set(NS3_CONTRIB_MODULES "" CACHE STRING "the module source directories, ;-separated")
option(NS3_CONTRIB_EXAMPLES "build each module's examples/ programs" ON)
if(NOT NS3_CONTRIB_PACKAGE OR NOT NS3_CONTRIB_MODULES)
  message(FATAL_ERROR "ns3-contrib: set NS3_CONTRIB_PACKAGE and NS3_CONTRIB_MODULES")
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)

find_package(ns3 CONFIG REQUIRED)
# ${lib<module>} for every installed ns-3 module, as inside an ns-3 tree.
get_directory_property(ns3_contrib_imported IMPORTED_TARGETS)
foreach(t IN LISTS ns3_contrib_imported)
  if(t MATCHES "^ns3::(.+)$")
    set(lib${CMAKE_MATCH_1} ${t})
  endif()
endforeach()
if(NOT TARGET ns3::core)
  message(FATAL_ERROR "ns3-contrib: the ns3 package has no ns3::core")
endif()
# NS3_VER (e.g. 3.49), from the installed core library's file name.
get_target_property(ns3_contrib_core ns3::core LOCATION)
if(NOT ns3_contrib_core MATCHES "libns(3\\.[0-9]+)-core")
  message(FATAL_ERROR "ns3-contrib: no ns-3 version in ${ns3_contrib_core}")
endif()
set(NS3_VER "${CMAKE_MATCH_1}")
set(ns3_contrib_suffix "")
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  set(ns3_contrib_suffix "-debug")
endif()
set(CMAKE_HEADER_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/include/ns3")
file(MAKE_DIRECTORY "${CMAKE_HEADER_OUTPUT_DIRECTORY}")
message(STATUS "ns3-contrib: ${NS3_CONTRIB_PACKAGE} on ns-${NS3_VER}")

function(build_lib)
  cmake_parse_arguments(BLIB "IGNORE_PCH;GENERATE_EXPORT_HEADER" "LIBNAME"
    "SOURCE_FILES;HEADER_FILES;PRIVATE_HEADER_FILES;DEPRECATED_HEADER_FILES;LIBRARIES_TO_LINK;TEST_SOURCES;MODULE_ENABLED_FEATURES"
    ${ARGN})
  set(target lib${BLIB_LIBNAME})
  add_library(${target} SHARED ${BLIB_SOURCE_FILES})
  set_target_properties(${target} PROPERTIES
    OUTPUT_NAME "ns${NS3_VER}-${BLIB_LIBNAME}${ns3_contrib_suffix}"
    EXPORT_NAME "${BLIB_LIBNAME}")
  target_link_libraries(${target} PUBLIC ${BLIB_LIBRARIES_TO_LINK})
  target_include_directories(${target} PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_BINARY_DIR}/include>"
    "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
  # ns3/<name>.h: a stub naming the header (unless the module wrote its own),
  # and the header itself installed under that name; then the umbrella header.
  set(umbrella "// ${BLIB_LIBNAME}-module.h, generated by the ns3 port's contrib build\n")
  foreach(h IN LISTS BLIB_HEADER_FILES BLIB_DEPRECATED_HEADER_FILES)
    get_filename_component(abs "${h}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    get_filename_component(name "${h}" NAME)
    if(NOT EXISTS "${CMAKE_HEADER_OUTPUT_DIRECTORY}/${name}")
      file(CONFIGURE OUTPUT "${CMAKE_HEADER_OUTPUT_DIRECTORY}/${name}"
           CONTENT "#include \"${abs}\"\n" @ONLY)
    endif()
    install(FILES "${abs}" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/ns3")
    string(APPEND umbrella "#include \"ns3/${name}\"\n")
  endforeach()
  file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/umbrella/${BLIB_LIBNAME}-module.h"
       CONTENT "${umbrella}" @ONLY)
  install(FILES "${CMAKE_BINARY_DIR}/umbrella/${BLIB_LIBNAME}-module.h"
          DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/ns3")
  install(TARGETS ${target} EXPORT ns3ExportTargets
          LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}"
          ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}"
          RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
  set(lib${BLIB_LIBNAME} ${target})
  set(lib${BLIB_LIBNAME} ${target} PARENT_SCOPE)
  if(NS3_CONTRIB_EXAMPLES AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/examples/CMakeLists.txt")
    add_subdirectory(examples)
  endif()
endfunction()

function(build_lib_example)
  cmake_parse_arguments(BLIB "IGNORE_PCH" "NAME" "SOURCE_FILES;HEADER_FILES;LIBRARIES_TO_LINK" ${ARGN})
  add_executable(${BLIB_NAME} ${BLIB_SOURCE_FILES})
  target_link_libraries(${BLIB_NAME} PRIVATE ${BLIB_LIBRARIES_TO_LINK})
  install(TARGETS ${BLIB_NAME} RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
endfunction()

foreach(dir IN LISTS NS3_CONTRIB_MODULES)
  get_filename_component(name "${dir}" NAME)
  add_subdirectory("${dir}" "${CMAKE_BINARY_DIR}/modules/${name}")
endforeach()

set(pkg "${NS3_CONTRIB_PACKAGE}")
install(EXPORT ns3ExportTargets FILE "${pkg}Targets.cmake" NAMESPACE ns3::
        DESTINATION "share/${pkg}")
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/${pkg}Config.cmake" CONTENT
"include(CMakeFindDependencyMacro)
find_dependency(ns3 CONFIG)
include(\"\${CMAKE_CURRENT_LIST_DIR}/${pkg}Targets.cmake\")
" @ONLY)
install(FILES "${CMAKE_BINARY_DIR}/${pkg}Config.cmake" DESTINATION "share/${pkg}")
